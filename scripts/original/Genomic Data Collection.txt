#Genomic data collection for Enterobacter strains

conda activate eh_typing
curl -o datasets 'https://ftp.ncbi.nlm.nih.gov/pub/datasets/command-line/v2/linux-amd64/datasets'
chmod +x datasets

# 1. Download metadata package (FASTA only)

datasets download genome taxon "Enterobacter" --assembly-level complete,chromosome --dehydrated --filename enterobacter_complete.zip

# 2. Unzip and rehydrate into dataset folder

unzip enterobacter_complete.zip -d enterobacter_dataset
datasets rehydrate --directory enterobacter_dataset/ --max-workers 16

#Fetch Enterobacter genome into an organized TSV file: Accession | Organism | Isolation Source | All Attributes

sudo apt install -y jq

./datasets summary genome taxon "Enterobacter" --assembly-level complete,chromosome --as-json-lines | jq -r '[.accession, .organism.organism_name, (.assembly_info.biosample.isolation_source // "NA"), ([.assembly_info.biosample.attributes[]? | .name + ":" + .value] | join("; ") // "NA")] | @tsv' > enterobacter_metadata.tsv

#Run MLST on all strains, this was used only to infer ST61 strains and not used in final results due some discrepancies in assigning "Cronobacter", --scheme was applied

mlst --threads 4 --scheme ecloacae enterobacter_dataset/ncbi_dataset/data/*/*.fna > mlst_results.tsv

# 2. Extract just the accession list for download
cut -f2 strain_mapping.tsv > accessions.txt

# 3. Download genomes from NCBI
echo "Downloading 28 genomes..."
/mnt/d/enterobacter_complete/datasets download genome accession --inputfile accessions.txt --filename ani_genomes.zip

# 4. Extract raw genome files
unzip -o ani_genomes.zip -d extracted_genomes/

# 5. Map and copy FASTA files with their exact Strain Names

mkdir -p fasta_files

while IFS=$'\t' read -r strain acc; do
    fna_file=$(ls extracted_genomes/ncbi_dataset/data/${acc}/*.fna 2>/dev/null)
    if [ -f "$fna_file" ]; then
        cp "$fna_file" "fasta_files/${strain}.fna"
        echo "Saved: fasta_files/${strain}.fna"
    else
        echo "Warning: Genome for ${strain} (${acc}) not found!"
    fi
done < strain_mapping.tsv

# 6. Clean up temporary working files
rm -rf ani_genomes.zip extracted_genomes/ strain_mapping.tsv accessions.txt
